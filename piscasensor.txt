  <script>
    const FIREBASE_URL = "https://led-status-65919-default-rtdb.asia-southeast1.firebasedatabase.app";

    // Função para alterar o status do LED no Firebase
    function setLed(val) {
      fetch(`${FIREBASE_URL}/led/led_status.json`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(val)
      })
      .then(res => res.json())
      .then(data => console.log("LED atualizado:", data))
      .catch(err => console.error("Erro ao alterar LED:", err));
    }

    // Função para ler o valor do sensor continuamente
    function lerSensor() {
      fetch(`${FIREBASE_URL}/sensor/distancia.json`)
      .then(res => res.json())
      .then(distancia => {
        if (distancia !== null && distancia !== undefined) {
          // Exemplo: Converte a distância (cm) em porcentagem de 0 a 100%
          // Ajuste os valores 10 (nível cheio) e 100 (nível vazio) de acordo com o seu projeto
          let maxDist = 100;
          let minDist = 10;
          let pct = Math.round(((maxDist - distancia) / (maxDist - minDist)) * 100);
          
          if (pct > 100) pct = 100;
          if (pct < 0) pct = 0;

          document.getElementById("sensor-val").innerText = `${pct}% (${distancia} cm)`;
        }
      })
      .catch(err => console.error("Erro ao ler sensor:", err));
    }

    // Atualiza a leitura do sensor a cada 1.5 segundos
    setInterval(lerSensor, 1500);
    lerSensor();
  </script>
